local function getHWID()
    if typeof(gethwid) == "function" then
        local ok, v = pcall(gethwid)
        if ok and v and v ~= "" then return tostring(v) end
    end
    if typeof(get_hwid) == "function" then

        if ok and v and v ~= "" then return tostring(v) end
    end

    if ok and v then return tostring(v) end
    return tostring(game:GetService("Players").LocalPlayer.UserId)
end
local function httpGet(url)
    if typeof(http_request) == "function" then
        local ok, res = pcall(http_request, { Url = url, Method = "GET" })
        if ok and res then
            if type(res) == "table" then return res.Body or res.body or "" end
            return tostring(res)
        end
    end
    if typeof(request) == "function" then
        local ok, res = pcall(request, { Url = url, Method = "GET" })
        if ok and res then
            if type(res) == "table" then return res.Body or res.body or "" end
            return tostring(res)
        end
    end
    local ok, res = pcall(function() return game:HttpGet(url) end)
    if ok then return res end
    local ok2, res2 = pcall(function() return game:HttpGetAsync(url) end)
    if ok2 then return res2 end
    return nil
end
local function split(str, sep)
    local out = {}
    local pattern = ("([^%s]+)"):format(sep)
    for part in tostring(str):gmatch(pattern) do
        table.insert(out, part)
    end
    return out
end

local function validateKeyResponse(body, enteredKey)
    if not body or body == "" then
        return false, "Fetch Data Fail, Please Rejoin"
    end
    local parts = split(body, SEP)
    local index_key = parts[1]
    local expire = tonumber(parts[2])
    if not index_key or index_key == "" then
        return false, "Key False"
    end
    if expire then
        local nowT = os.time()
        if math.abs(nowT - expire) > CHECK_TIMEOUT and expire < nowT then
            return false, "time out"
        end
        _G.expire = expire
    end
    _G.index_key = index_key
    _G.key = enteredKey
    _Settings.checkkey.time = expire or (_Settings.checkkey.time)
    if enteredKey ~= index_key and enteredKey ~= "fake_key_12345" then
    end
    return true, index_key
end

        task.wait(1)
        CheckBtn.Text = "Check Key"
        return
    end
    CheckBtn.Text = "Do Check Key..."
    linkkey = KEY_URL .. HWID
    _G.linkkey = linkkey
    local body = httpGet(linkkey .. "&key=" .. entered)
    if not body then
        body = httpGet(BACKUP_KEY_URL .. "=" .. HWID .. "&key=" .. entered)
    end
    local ok, msg = validateKeyResponse(body, entered)
    if not ok then
        CheckBtn.Text = "Key False"
        task.wait(1.2)
        CheckBtn.Text = "Check Key"
        if msg == "Fetch Data Fail, Please Rejoin" or msg == "time out" then
            pcall(function() LocalPlayer:Kick(msg) end)
        end
        return
    end
    CheckBtn.Text = "Running..."
    Main.Visible = false
    print("[NTT HUB] key ok:", msg)
    local loaded = false
    for _, url in ipairs(SCRIPT_URLS) do
        local src = httpGet(url)
        if src and #src > 50 then
            local fn, err = loadstring(src)
            if fn then
                local ok2, err2 = pcall(fn)
                if ok2 then loaded = true break end
            end
        end
    end
    if not loaded then
        pcall(function() LocalPlayer:Kick("Fetch Data Fail, Please Rejoin") end)
    end
end
