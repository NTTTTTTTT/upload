local HttpService = game:GetService("HttpService")

local function getDiscordInvite(inviteCode, field)
    if not inviteCode or inviteCode == "" then
        return nil
    end

    local requestFn = (http and http.request) or request or http_request
    if not requestFn then
        return nil
    end

    local ok, response = pcall(requestFn, {
        Url = string.format("https://discord.com/api/v9/invites/%s?with_counts=true&with_expiration=true", inviteCode),
        Method = "GET",
        Headers = { ["User-Agent"] = "Mozilla/5.0" },
    })

    if not ok or not response or response.StatusCode ~= 200 then
        return nil
    end

    local decodeOk, data = pcall(HttpService.JSONDecode, HttpService, response.Body)
    if not decodeOk or not data or not data.guild then
        return nil
    end

    local guild = data.guild
    local info = {
        Name = guild.name,
        Description = guild.description,
        GuildID = guild.id,
        InviteCode = data.code,
        MemberCount = data.approximate_member_count,
        OnlineCount = data.approximate_presence_count,
        Icon = guild.icon,
        IconURL = guild.icon and string.format("https://cdn.discordapp.com/icons/%s/%s.png?size=1024", guild.id, guild.icon) or nil,
        NSFW = guild.nsfw or false,
        CreatedAt = data.created_at,
    }

    if field == nil or field == "All" then
        return info
    end

    return info[field]
end

local function checkKey(key)
    local keyToCheck = key or scriptKey
    local result = Luarmor.check_key(keyToCheck)

    if result.code == "KEY_VALID" then
        if result.data.note ~= "Ad Reward" then
            _G.Premium = true
            Status = "Premium"
            print("Premium Mode: true")
            queue_on_teleport("_G.Premium = true")
        else
            _G.Normal = true
            print("Premium Mode: false")
        end

        scriptKey = keyToCheck
        script_key = keyToCheck
        writefile("KeySystem/Key.txt", keyToCheck)

        Fluent:Destroy()
        Luarmor.load_script()
        return
    end

    if result.code and result.code:find("KEY_") then
        local messages = {
            KEY_HWID_LOCKED = "Key is locked to another HWID",
            KEY_INCORRECT = "Key is incorrect",
            KEY_INVALID = "Key is invalid",
        }

        notify(
            "Key Check Failed",
            messages[result.code] or result.message or "Unknown key error"
        )
        return
    end

    Players.LocalPlayer:Kick(
        "Key check failed: "
            .. tostring(result.message)
            .. " Code: "
            .. tostring(result.code)
    )
end
