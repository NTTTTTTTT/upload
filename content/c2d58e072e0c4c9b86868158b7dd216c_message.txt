local Functions, Events
do
    local ok, rs = pcall(function() return game:GetService("ReplicatedStorage") end)
    if ok and rs then
        local chest = rs:FindFirstChild("Chest")
        if chest then
            local rem = chest:FindFirstChild("Remotes")
            if rem then
                Functions = rem:FindFirstChild("Functions")
                Events = rem:FindFirstChild("Events")
            end
        end
    end
end

local function invoke(name, ...)
    if not Functions then return false, "no Functions" end
    local r = Functions:FindFirstChild(name)
    if not r then return false, "missing " .. tostring(name) end
    local args = { ... }
    local res = { pcall(function() return r:InvokeServer(unpack(args)) end) }
    if res[1] then return true, res[2] end
    return false, res[2]
end

local function fire(name, ...)
    if not Events then return false, "no Events" end
    local r = Events:FindFirstChild(name, true)
    if not r then return false, "missing " .. tostring(name) end
    local args = { ... }
    local ok, err = pcall(function() r:FireServer(unpack(args)) end)
    return ok, err
end

local function getHRP()
    local c = LocalPlayer.Character
    return c and c:FindFirstChild("HumanoidRootPart") or nil
end

local function currentToolName()
    local c = LocalPlayer.Character
    if c then local t = c:FindFirstChildOfClass("Tool") if t then return t.Name end end
    local bp = LocalPlayer:FindFirstChild("Backpack")
    if bp then local t = bp:FindFirstChildOfClass("Tool") if t then return t.Name end end
    return nil
end

local function equipToolMatch(pat)
    local bp = LocalPlayer:FindFirstChild("Backpack")
    local c = LocalPlayer.Character
    if c and c:FindFirstChildOfClass("Tool") then return true end
    if bp then for _, t in ipairs(bp:GetChildren()) do
        if t:IsA("Tool") and t.Name:lower():find(pat:lower()) then
            pcall(function()
                local h = c and c:FindFirstChild("Humanoid")
                if h then h:EquipTool(t) end
            end)
            return true
        end
    end end
    return false
end

local function swordM1()
    local t = currentToolName()
    if not t then return false end
    local ok = invoke("SkillAction", "SW_" .. t .. "_M1")
    return ok
end

local function combatFolders(sea)
    local out = {}
    local function add(p) if p then out[#out + 1] = p end end
    if not sea then
        local m = workspace:FindFirstChild("Monster")
        if m then add(m:FindFirstChild("Mon")) add(m:FindFirstChild("Boss")) end
    else

        add(workspace:FindFirstChild("SeaMonster"))
        add(workspace:FindFirstChild("GhostMonster"))
        local ships = workspace:FindFirstChild("Ships")
        if ships then for _, s in ipairs(ships:GetChildren()) do add(s) end end
    end
    return out
end

local function mobAlive(m)
    if not m or not m:IsA("Model") then return false end
    local h = m:FindFirstChild("Humanoid")
    local hrp = m:FindFirstChild("HumanoidRootPart")
    if not h or not hrp then return false end
    local ok, hp = pcall(function() return h.Health end)
    return ok and hp > 0
end

local function baseName(n) return tostring(n):gsub("%s*%[Lv%.?%s*%d+%]%s*$", "") end

local function questTarget(questName)
    return tostring(questName):match("Kill%s+%d+%s+(.+)") or tostring(questName)
end

local function matchScore(a, b)
    a, b = a:lower(), b:lower()
    if a == b then return 100 end
    local function w(s) local t = {} for x in s:gmatch("%S+") do t[#t + 1] = x end return t end
    local aw, bw, s = w(a), w(b), 0
    for i = 1, math.min(2, #aw, #bw) do
        local x, y = aw[i], bw[i]
        if x == y or x == y .. "s" or y == x .. "s" or x:sub(1, 5) == y:sub(1, 5) then s = s + 10 end
    end
    return s
end

local function nearestMob(target, sea, bossOnly)
    local hrp = getHRP()
    if not hrp then return nil end
    local best, bestD = nil, math.huge
    for _, folder in ipairs(combatFolders(sea)) do
        local guard = folder and folder:GetChildren()
        if guard then for _, m in ipairs(guard) do
            if mobAlive(m) then
                local base = baseName(m.Name)
                local okMatch = true
                if target and target ~= "" then okMatch = matchScore(base, target) >= 10 end
                if okMatch and bossOnly then
                    okMatch = folder.Name == "Boss" or matchScore(base, target or "") >= 100
                end
                if okMatch then
                    local mhrp = m:FindFirstChild("HumanoidRootPart")
                    if mhrp then
                        local d = (mhrp.Position - hrp.Position).Magnitude
                        if d < bestD then best, bestD = m, d end
                    end
                end
            end
        end end
    end
    return best
end

local function gotoMob(m)
    local hrp = getHRP()
    local mhrp = m and m:FindFirstChild("HumanoidRootPart")
    if not hrp or not mhrp then return false end
    pcall(function()
        hrp.CFrame = mhrp.CFrame * CFrame.new(0, State.DistanceAbove, 0) * CFrame.Angles(math.rad(-90), 0, 0)
        hrp.Velocity = Vector3.new(0, 0, 0)
    end)
    return true
end

local farmToken, bossToken, seaToken, miscToken = 0, 0, 0, 0

local function farmLoop()
    farmToken = farmToken + 1 local my = farmToken
    State.Status = "farm start"
    invoke("Quest", "take", State.QuestName)
    local lastRetake = 0
    while State.AutoFarm and my == farmToken do
        local ok = pcall(function()
            local target = questTarget(State.QuestName)
            local mob = nearestMob(target, false, false)
            if not mob then
                if os.clock() - lastRetake > 2 then
                    lastRetake = os.clock()
                    invoke("Quest", "take", State.QuestName)
                    State.Status = "waiting spawn: " .. target
                end
                task.wait(0.5) return
            end
            gotoMob(mob) swordM1()
            State.Status = "farming: " .. mob.Name
        end)
        if not ok then task.wait(0.5) end
        task.wait(State.AttackDelay)
    end
    State.Status = "idle"
end

local function bossLoop()
    bossToken = bossToken + 1 local my = bossToken
    while State.AutoKillBoss and my == bossToken do
        pcall(function()
            local mob = nil
            if State.BossName ~= "" then
                for _, f in ipairs(combatFolders(false)) do
                    for _, m in ipairs(f:GetChildren()) do
                        if mobAlive(m) and baseName(m.Name):lower() == State.BossName:lower() then mob = m break end
                    end
                    if mob then break end
                end
            else
                mob = nearestMob(nil, false, true)
            end
            if mob then gotoMob(mob) swordM1() State.Status = "boss: " .. mob.Name
            else State.Status = "boss waiting spawn" task.wait(1) end
        end)
        task.wait(State.AttackDelay)
    end
    State.Status = "idle"
end

local function seaLoop(kind)
    seaToken = seaToken + 1 local my = seaToken
    while State.AutoSea and my == seaToken do
        pcall(function()
            local mob = nearestMob(kind ~= "" and kind or nil, true, false)
            if mob then gotoMob(mob) swordM1() State.Status = "sea: " .. mob.Name
            else State.Status = "sea waiting: " .. (kind ~= "" and kind or "any") task.wait(1) end
        end)
        task.wait(State.AttackDelay)
    end
    State.Status = "idle"
end

local function shopLoop()
    miscToken = miscToken + 1 local my = miscToken
    while (State.AutoBuyChest or State.AutoBuyKey) and my == miscToken do
        pcall(function()
            if State.AutoBuyKey then
                local ok = invoke("BuyKey")
                State.Status = "BuyKey " .. (ok and "sent" or "missing?")
            end
            if State.AutoBuyChest then
                local ok1 = invoke("BuyFruitStock")
                local ok2 = fire("BuyChest")
                State.Status = "chest try BuyFruitStock=" .. tostring(ok1) .. " BuyChest=" .. tostring(ok2)
            end
        end)
        task.wait(3)
    end
    State.Status = "idle"
end

local function fishLoop()
    local my = miscToken + 1 miscToken = my
    State.AutoFish = true
    while State.AutoFish and miscToken == my do
        pcall(function()
            if equipToolMatch("rod") or equipToolMatch("fish") then
                local c = LocalPlayer.Character
                local t = c and c:FindFirstChildOfClass("Tool")
                if t then pcall(function() t:Activate() end) end
                State.Status = "fishing cast..."
            else
                State.Status = "[TPL] no rod tool found"
            end
        end)
        task.wait(4)
    end
    State.Status = "idle"
end

local function dungeonOnce()

end

local function conquestLoop()
    local my = miscToken + 1 miscToken = my
    State.AutoConquest = true
    while State.AutoConquest and miscToken == my do
        pcall(function()
            local f = workspace:FindFirstChild("FlagConquest")
            if f and #f:GetChildren() > 0 then
                local hrp = getHRP()
                for _, p in ipairs(f:GetChildren()) do
                    if not State.AutoConquest then break end
                    local cf = p:IsA("BasePart") and p.CFrame or (p:FindFirstChildWhichIsA("BasePart") and p:FindFirstChildWhichIsA("BasePart").CFrame)
                    if cf and hrp then pcall(function() hrp.CFrame = cf + Vector3.new(0, 8, 0) end) end
                    State.Status = "conquest: " .. p.Name
                    task.wait(2)
                end
            else
                State.Status = "[TPL] no FlagConquest points"
                task.wait(3)
            end
        end)
        task.wait(1)
    end
    State.Status = "idle"
end

local function allyLoop()

            State.Status = "ally accept sent=" .. tostring(ok)
        end)

    end
    State.Status = "idle"
end

local function bringFruit()
    local hrp = getHRP()
    local f = workspace:FindFirstChild("AllDroppedFruit")
    if not hrp or not f then return 0 end
    local n = 0
    for _, m in ipairs(f:GetChildren()) do
        local p = m:IsA("BasePart") and m or m:FindFirstChildWhichIsA("BasePart")
        if p then pcall(function() p.CFrame = hrp.CFrame end) n = n + 1 end
    end
    return n
end
